function collapse(arr){
  const best={};
  for(const s of (arr||[])){ if(!streamOf(s)||s.lastcheckok===0||dead.has(s.stationuuid))continue;
    const k=stem(s.name)||s.stationuuid;
    const held=best[k];
    if(!held){ best[k]=s; continue; }
    const win=((s.bitrate||0)>(held.bitrate||0))?s:held, lose=(win===s)?held:s;
    if(!win.favicon&&lose.favicon)win.favicon=lose.favicon;
    if(!win.homepage&&lose.homepage)win.homepage=lose.homepage;
    if(!win.tags&&lose.tags)win.tags=lose.tags;
    if(!win.state&&lose.state)win.state=lose.state;
    if((lose.votes||0)>(win.votes||0))win.votes=lose.votes;
    best[k]=win; }
  return Object.values(best).map(reg);
}
function scoreBy(list,rank,accepted){
  const focus=s=>{ if(!accepted||!accepted.size)return 1;
    const ts=tagsOf(s); if(!ts.length)return 1;
    const hits=ts.filter(t=>accepted.has(t)).length;
    if(!hits)return 0.35;
    return 1+1.6*(hits/Math.sqrt(ts.length));
  };
  const spam=s=>{let m=1; if(tagsOf(s).length>25)m*=0.3;
    if(/^[^0-9a-zA-ZÀ-ɏ]/.test((s.name||'').trim()))m*=0.2; return m;};
  const f={
    best:s=>Math.log10((s.votes||0)+2)*spam(s),
    trending:s=>((s.clicktrend||0)*100+(s.clickcount||0))*spam(s),
    underrated:s=>((s.votes||0)>=5?(s.votes||0)/((s.clickcount||0)+3):0)*spam(s),
    hifi:s=>((s.bitrate||0)/64+Math.log10((s.votes||0)+2))*spam(s),
    random:()=>Math.random(),
  }[rank]||(s=>(s.votes||0)*spam(s));
  return list.slice().sort((a,b)=>f(b)*focus(b)-f(a)*focus(a));
}

async function fetchPool({tags,cc,state,rank='best',limit=200}){
  const qs=(tags&&tags.length)?tags.slice(0,4):[null];
  const rs=await Promise.all(qs.map(t=>{
    let p='/stations/search?hidebroken=true&order=votes&reverse=true&limit='+limit;
    if(t)p+='&tag='+enc(t)+'&tagExact=true';
    if(cc)p+='&countrycode='+cc;
    if(state)p+='&state='+enc(state);
    return api(p).catch(()=>[]);
  }));
  const seen=new Set(),merged=[];
  for(const s of [].concat(...rs)){ if(!s.stationuuid||seen.has(s.stationuuid))continue;
    seen.add(s.stationuuid);merged.push(s); }
  return scoreBy(collapse(merged),rank);
}
async function doorwayPool({genre,cc,state,rank='best'}){
  const tags=genre?GENRE[genre].tags:null;
  let pool=await fetchPool({tags,cc,state,rank});
  let note='';
  if(pool.length<5&&state){
    const p2=await fetchPool({tags,cc,rank});
    if(p2.length>pool.length){pool=p2;note='Only '+pool.length+' in '+state+' — widened to '+(CNAME[cc]?CNAME[cc].name:'the country');}
  }
  if(pool.length<5&&cc){
    const region=CNAME[cc]?CNAME[cc].region:null;
    if(region){
      const nb=COUNTRIES.filter(c=>c[3]===region&&c[0]!==cc).slice(0,6).map(c=>c[0]);
      const rs=await Promise.all(nb.map(c=>fetchPool({tags,cc:c,rank,limit:60}).catch(()=>[])));
      const wide=scoreBy(collapse([].concat(...rs)),rank);
      if(wide.length>=3){pool=pool.concat(wide);note='Thin in '+(CNAME[cc]?CNAME[cc].name:'that country')+' — opened out to '+region;}
    }
  }
  if(pool.length<3&&genre){
    const w=await fetchPool({tags,rank});
    if(w.length){pool=w;note='Nothing local — '+genre+' worldwide instead';}
  }
  return {pool,note};
}

function parseQuery(raw){
  let q=' '+raw.toLowerCase().replace(/[^a-z0-9&\- ]/g,' ').replace(/\s+/g,' ')+' ';
  const out={genre:null,matched:null,cc:null,rank:null,rest:[],text:raw.trim()};
  const eat=phrase=>{const p=' '+phrase+' '; if(q.includes(p)){q=q.replace(p,' ');return true;} return false;};
  for(const r of RANKINGS)
    for(const w of r.words.slice().sort((a,b)=>b.length-a.length))
      if(!out.rank&&eat(w))out.rank=r.k;
  const places=[...Object.keys(CC),...Object.keys(DEMONYM)].sort((a,b)=>b.length-a.length);
  for(const p of places){ if(out.cc)break;
    if(eat(p))out.cc=CC[p]||DEMONYM[p]; }
  const gs=Object.keys(GENRE_LOOKUP).sort((a,b)=>b.length-a.length);
  for(const g of gs){ if(out.genre)break;
    if(eat(g)){ out.genre=GENRE_LOOKUP[g]; out.matched=g; } }
  out.rest=q.trim().split(' ').filter(w=>w&&w.length>1&&!['in','the','on','of','from','radio','music','station','stations','me','a','some','play'].includes(w));
  return out;
}
function describe(p){

  if(p.rank)bits.push(chip(esc(RANKINGS.find(r=>r.k===p.rank).label)));
  if(p.matched||p.genre){
    const m=p.matched||p.genre;
    const fam=(p.genre&&fold(p.genre)!==fold(m))?' <i>· '+esc(p.genre)+'</i>':'';
    bits.push(chip(esc(m)+fam));

  if(p.cc)bits.push(chip((flagOf(p.cc)||'')+' '+esc(countryName(p.cc))));
  if(p.place)bits.push(chip('\u{1F4CD} '+esc(p.place)));
  if(p.rest.length)bits.push(chip('“'+esc(p.rest.join(' '))+'”'));
  return bits.join('');
}

function seekBy(sec){
  if(!cur)return;
  try{
    const sk=au.seekable;
    if(!sk||!sk.length){ flash('This stream cannot be rewound'); return; }
    const lo=sk.start(0), hi=sk.end(sk.length-1);
    if(hi-lo<2){ flash('Nothing buffered to skip through yet'); return; }
    const want=(au.currentTime||0)+sec;
    const t=Math.max(lo,Math.min(want,hi));
    if(Math.abs(t-(au.currentTime||0))<0.5){
      flash(sec<0?'Already at the start of the buffer':'Already live'); return;
    }
    au.currentTime=t;
    flash(sec<0?'Back 10 seconds':'Forward 10 seconds');
  }catch(e){ flash('This stream cannot be rewound'); }
}

function hopTo(i){
  clearScanTimers(); if(i<0)i=0;
  if(i>=scan.queue.length){lockIn();return;}
