let artObs=null;
function hydrateArt(){
  if(!('IntersectionObserver' in window))return;
  if(!artObs)artObs=new IntersectionObserver(es=>{
    es.forEach(en=>{
      if(!en.isIntersecting)return;
      const el=en.target; artObs.unobserve(el);
      const hp=el.dataset.arthp; if(!hp)return;
      fetch('/art?u='+enc(hp),{cache:'force-cache'}).then(r=>r.json()).then(j=>{
        if(!j||!j.art||!el.parentNode)return;
        const im=new Image(); im.className=el.className; im.loading='lazy';
        im.onload=()=>{ if(el.parentNode)el.replaceWith(im); };
        im.src=proxied(j.art);
      }).catch(()=>{});
    });
  },{rootMargin:'200px'});
  document.querySelectorAll('[data-arthp]').forEach(el=>artObs.observe(el));
}

const NAME_CACHE={};
function niceName(s){
  const raw=(s&&s.name)||'';
  if(NAME_CACHE[raw]!==undefined)return NAME_CACHE[raw];
  let n=raw.replace(/[​-‏‪-‮﻿�]/g,'').trim();
  n=n.replace(/[\(\[][^)\]]*\b(mp3|aac\+?|flac|ogg|opus|m4a|kbps|\d{2,3}\s*k|hd|hq|stereo|mono)\b[^)\]]*[\)\]]/gi,' ');
  n=n.replace(/\s*[-–—|]?\s*\b\d{2,3}\s*k(bps)?\b/gi,' ');
  n=n.replace(/\s*[-–—|]?\s*\b(mp3|aac\+|aac|flac|ogg|opus|m4a)\b/gi,' ');
  n=n.replace(/\s*[-–—]\s*(hd|hq|sd|live|stream|online|official)\s*$/i,'');
  n=n.replace(/\s+\b(hd|hq)\b\s*$/i,'');
  n=n.replace(/\s{2,}/g,' ').replace(/[\s\-–—|·,]+$/,'').replace(/^[\s\-–—|·,]+/,'').trim();
  NAME_CACHE[raw]=n||raw;
  return NAME_CACHE[raw];
}

function buildStyles(){
  const seen=new Set(),out=[];
  VOCAB.tags.forEach(([name,count])=>{
    const f=fold(name);
    if(!f||seen.has(f)||count<10)return;
    if(JUNK_TAG.test(name)||GEO_WORDS.test(f)||LANG_WORDS.test(f))return;
    if(/(m[eé]xico|ciudad de|valle de|provincia|departamento|estado de|condado)/.test(f))return;
    if(f.split(' ').length>3)return;
    if(/\d{2,}[.,]\d|^\d+$|http|\.com|@|^\W/.test(f))return;
    const vague=/\b(music|música|musica|radio|local|regional|popular|estacion|estación|en español|en ingles|en inglés|variety|varied|mixed|en vivo|programas|juvenil|informacion|información|information)\b/.test(f);
    const fam=familyOf(f);
    if(!fam&&!ENGLISH_EXTRA.test(f))return;
    if(/[ñáéíóúàâãçêôõü]|ção|ción/.test(f))return;
    seen.add(f);
    out.push({name,count,family:fam,tier:BROAD.has(f)?2:(vague?1:0)});
  });
  STYLES=out.sort((a,b)=>a.tier-b.tier||b.count-a.count);
}
function familyOf(f){
  for(const F of TAX)
    for(const [nm,tags] of F.g)
      for(const t of tags.concat([nm]))
        if(new RegExp('(^|\\s)'+fold(t).replace(/[.*+?^${}()|[\]\\]/g,'\\$&')+'($|\\s)').test(f))
          return F.f;
  return null;
}

    VOCAB.tags.forEach(([name,count])=>{
      const f=fold(name);
      if(!words.some(w=>new RegExp('(^|\\s)'+w.replace(/[.*+?^${}()|[\]\\]/g,'\\$&')+'($|\\s)').test(f)))return;
      add(name,count);
    });
    (EXTRA_SUBS[g.name]||[]).forEach(t=>{
      const hit=VOCAB.tags.find(x=>fold(x[0])===fold(t)); if(hit)add(hit[0],hit[1]);
    });
    SUBS[g.name]=kids.sort((a,b)=>b[1]-a[1]).slice(0,44);
  });
}

function mixCountLabel(){
  const b=[];
  if(mix.genres.length)b.push(mix.genres.length+' genre'+(mix.genres.length>1?'s':''));
  if(mix.countries.length)b.push(mix.countries.length+' place'+(mix.countries.length>1?'s':''));
  if(mix.regions.length)b.push(mix.regions.length+' region'+(mix.regions.length>1?'s':''));
  if(mix.languages.length)b.push(mix.languages.length+' language'+(mix.languages.length>1?'s':''));
  return b.join(', ')||'everything';
}
function mixLabel(){
  const b=[];
  if(mix.ranking)b.push(RANKINGS.find(r=>r.k===mix.ranking).label);
  if(mix.genres.length)b.push(mix.genres.map(g=>g.name).join(' + '));
  if(mix.regions.length)b.push(mix.regions.join(' + '));
  else if(mix.countries.length)b.push(mix.countries.map(countryName).join(' + '));
  if(mix.languages.length)b.push('in '+mix.languages.join('/'));
  return b.join(' · ')||'Everything';
}
function mixEstimate(){
  const g=mix.genres.reduce((a,x)=>a+genreCount(x),0);
  const c=mix.countries.reduce((a,x)=>a+ccCount(x),0);
  if(g&&c)return Math.min(g,c);
  return g||c||0;
}
async function mixPool(m,floor=8){
  const seen=new Set();let out=[];
  const accepted=new Set(m.genres.flatMap(g=>g.tags.map(t=>t.toLowerCase())));
  const banned=new Set(m.excluded.map(t=>t.toLowerCase()));
  async function harvest(gs,ccs,state){
    const tagSets=gs.length?gs.map(g=>g.tags):[[null]];
    const places=ccs.length?ccs:[null];
    for(const place of places){
      for(const tags of tagSets){
        let added=0;
        for(const tag of tags){
          let p='/stations/search?hidebroken=true&order=votes&reverse=true&limit=80';
          if(tag)p+='&tag='+enc(tag)+'&tagExact=true';
          if(place)p+='&countrycode='+place;
          if(state)p+='&state='+enc(state);
          if(m.languages[0])p+='&language='+enc(m.languages[0]);
          const batch=await api(p).catch(()=>[]);
          const alive=batch.filter(s=>s.stationuuid&&!seen.has(s.stationuuid)&&s.lastcheckok!==0
            &&streamOf(s)&&!dead.has(s.stationuuid)
            &&(!m.minBitrate||(s.bitrate||0)>=m.minBitrate)
            &&!(banned.size&&tagsOf(s).some(t=>banned.has(t))));
          const strict=accepted.size?alive.filter(s=>tagsOf(s).some(t=>accepted.has(t))):alive;
          for(const s of (strict.length?strict:alive)){seen.add(s.stationuuid);out.push(s);added++;}
          if(added>=15)break;
          if(out.length>=220)return;
        }
      }
    }
  }
  await harvest(m.genres,m.countries,m.regions[0]||null);
  let note='';
  if(out.length<floor&&m.regions.length){
    await harvest(m.genres,m.countries,null);
    note='Thin in '+m.regions[0]+' — widened to the country';
  }
  if(out.length<floor&&m.countries.length){
    const seed=m.countries[0],reg=CNAME[seed]?CNAME[seed].region:null;
    if(reg){
      const near=COUNTRIES.filter(c=>c[3]===reg&&!m.countries.includes(c[0])).slice(0,6).map(c=>c[0]);
      await harvest(m.genres,near,null);
      if(out.length)note='Thin in '+countryName(seed)+' — opened out to '+reg;
    }
  }
  if(out.length<floor&&m.genres.length){
    await harvest(m.genres,[],null);
    note=(m.countries.length?'Nothing local — ':'')+mix.genres.map(g=>g.name).join(' + ')+' worldwide';
  }
  return {pool:scoreBy(collapse(out),m.ranking||'best',accepted),note};
}
async function playMix(startIdx=0){
  if(mixEmpty())return;
  const snap=JSON.parse(JSON.stringify({genres:mix.genres,countries:mix.countries,
    regions:mix.regions,languages:mix.languages,excluded:mix.excluded,ranking:mix.ranking}));
  const btn=$('deckPlay'); if(btn)btn.innerHTML='<span class="spin"></span>';
  let r;
  try{r=await mixPool(snap);}catch(e){if(btn)btn.textContent='▶';return;}
  if(btn)btn.textContent='▶';
  if(!r.pool.length)return;
  lockGenre=snap.genres.length?snap.genres[0].name:null;
  lockCC=snap.countries[0]||null;
  doorwayLabel=mixLabel()+(r.note?' · widened':'');
  if(r.note){ notice(r.note); setTimeout(()=>{if($('notice').textContent===r.note)notice('');},5000); }
  forwardQueue=varyOrder(freshOnly(r.pool));
  if(forwardQueue.length<3)forwardQueue=varyOrder(r.pool.slice());
  const start=(snap.ranking==='random')?Math.floor(Math.random()*Math.min(forwardQueue.length,12)):startIdx;
  playStation(forwardQueue.splice(start,1)[0],{keepQueue:true});
  lastMix=snap;
  goPane(0);renderLocks();
}
