function chipsFor(){
  const out=[];
  if(mix.ranking)out.push(['rank',mix.ranking,RANKINGS.find(r=>r.k===mix.ranking).label]);
  if(mix.minBitrate)out.push(['br',String(mix.minBitrate),mix.minBitrate+'k+']);
  mix.genres.forEach(g=>out.push(['genre',g.name,g.name]));
  mix.countries.forEach(c=>out.push(['country',c,flagOf(c)+' '+countryName(c)]));
  mix.regions.forEach(r=>out.push(['region',r,'📍 '+r]));
  mix.languages.forEach(l=>out.push(['lang',l,'in '+l]));
  mix.excluded.forEach(x=>out.push(['excl',x,'− '+x]));
  return out;
}

function adopt(p){
  if(p.genre&&!mix.genres.some(x=>x.name===p.genre))mix.genres.push(GENRE[p.genre]);
  if(p.cc&&!mix.countries.includes(p.cc))mix.countries.push(p.cc);
  if(p.rank)mix.ranking=p.rank;
  if(p.rest.length){
    const t=p.rest.join(' ');
    const hit=VOCAB.tags.find(x=>fold(x[0])===fold(t));
    if(hit&&!mix.genres.some(g=>g.name===hit[0]))mix.genres.push({name:hit[0],tags:[hit[0]],family:'Tag',emoji:'🎚'});
  }
  afterPick();
}
function renderSuggestions(){
  const box=$('sugg'); if(!box)return;
  const f=fold(indexFilter);
  if(f.length<2){box.innerHTML='';box.classList.remove('on');return;}
  const tier=n=>{const x=fold(n);if(x.startsWith(f))return 0;
    if(x.split(' ').some(w=>w.startsWith(f)))return 1; if(x.includes(f))return 2; return null;};
  const score=(t,c)=>t*1e5-Math.min(99999,c||0);
  const out=[];
  const p=parseQuery(indexFilter);
  if((p.genre?1:0)+(p.cc?1:0)+(p.rank?1:0)>=1)
    out.push([-1e6,'✦ '+describe(p),()=>{adopt(p);},true]);
  Object.values(GENRE).forEach(g=>{const t=tier(g.name);
    if(t!==null&&!mix.genres.some(x=>x.name===g.name))
      out.push([score(t,genreCount(g))-6e4,
        g.name+'  '+(genreCount(g)||''),()=>mixToggleGenre(g)]);});
  TAX.forEach(F=>{const t=tier(F.f);
    if(t!==null)out.push([-5e5,F.e+' '+F.f+' · all',()=>mixAddFamily(F)]);});
  VOCAB.tags.forEach(([name,count])=>{const t=tier(name);
    if(t===null||GENRE_LOOKUP[name]||mix.genres.some(g=>g.name===name))return;
    out.push([t*1e5-Math.min(9999,count||0),name+'  '+count,
      ()=>mixToggleGenre({name,tags:[name],family:'Tag',emoji:'🎚'})]);});
  VOCAB.languages.forEach(([name,count])=>{const t=tier(name);
    if(t===null||mix.languages.includes(name))return;
    out.push([score(t,count)+2e4,'in '+name+'  '+count,()=>mixToggleLang(name)]);});
  VOCAB.countries.forEach(([cc,count])=>{
    const t=Math.min(tier(countryName(cc))??9,
      Object.entries(DEMONYM).filter(([,c])=>c===cc).map(([w])=>tier(w)??9).concat(9)[0]);
    if(t>2||mix.countries.includes(cc))return;
    out.push([score(t,count),flagOf(cc)+' '+countryName(cc)+'  '+count,()=>mixToggleCountry(cc)]);});
  (regionCache[mix.countries[0]]||[]).forEach(([r,count])=>{const t=tier(r);
    if(t===null||mix.regions.includes(r))return;
    out.push([score(t,count)+5e3,'📍 '+r+'  '+count,()=>mixToggleRegion(r)]);});
  const list=out.sort((a,b)=>a[0]-b[0]||a[1].length-b[1].length).slice(0,8);
  suggActions=list.map(x=>x[2]);
  box.classList.add('on');

    +(x[3]?x[1]:esc(x[1]))+'</button>').join('');
  townSuggest((p.rest&&p.rest.length)?p.rest.join(' '):indexFilter, box);
  nameSuggest(indexFilter, box);
}

const geoTown={};
async function townSearch(term){
  const key=fold(term); if(!key)return null;
  if(geoTown[key]!==undefined)return geoTown[key];
  let hit=null;
  try{
    const g=await fetch('https://geocoding-api.open-meteo.com/v1/search?count=1&language=en&name='+enc(term))
      .then(r=>r.json());
    hit=(g.results||[])[0]||null;
  }catch(e){}
  if(!hit){ geoTown[key]=null; return null; }
  let rows=[];
  for(const km of [50,150,400]){
    try{
      rows=await api('/stations/search?hidebroken=true&limit=80&order=votes&reverse=true'
        +'&geo_lat='+hit.latitude+'&geo_long='+hit.longitude+'&geo_distance='+(km*1000));
    }catch(e){ rows=[]; }
    if(rows.length>=3)break;
  }
  const out={place:hit.name,country:hit.country_code||'',admin:hit.admin1||'',
             lat:hit.latitude,lon:hit.longitude,rows:rows||[]};
  geoTown[key]=out; return out;
}

async function loadStations(scope,tags){
  ixLoading=true; ixScope=scope; ixPool=null;
  const chunks=[]; for(let i=0;i<Math.min(tags.length,12);i+=4)chunks.push(tags.slice(i,i+4));
  let pool=[];
  try{
    const rs=await Promise.all(chunks.map(t=>fetchPool({tags:t,rank:ixSort}).catch(()=>[])));
    const seen=new Set();
    for(const s of [].concat(...rs)){
      if(!s.stationuuid||seen.has(s.stationuuid))continue;
      seen.add(s.stationuuid); pool.push(s);
    }
    if(chunks.length>1)pool=scoreBy(pool,ixSort);
  }catch(e){}
  ixLoading=false;
  if(ixScope!==scope)return;
  ixPool=pool.filter(x=>streamOf(x));
  listDoor={pool:ixPool,genre:GENRE[scope]?scope:null,cc:null,label:scope};
  renderIndex();
}

async function loadRegions(cc){
  if(regionCache[cc])return;
  regionCache[cc]=null;
  const list=await api('/stations/search?countrycode='+cc+'&hidebroken=true&order=votes&reverse=true&limit=400').catch(()=>[]);
  const tally={};
  (list||[]).forEach(s=>{const r=(s.state||'').trim();
    if(!r||r.length<2||/^https?:|@|^-|unknown|n\/a/i.test(r))return;
    const k=r.replace(/[.,]+$/,'').trim(); tally[k]=(tally[k]||0)+1;});
  regionCache[cc]=Object.entries(tally).filter(x=>x[1]>=2).sort((a,b)=>b[1]-a[1]);
  renderIndex();
}

function buildDoorways(limit=12){
  const h=new Date().getHours();
  const moodFor=h<5?['Ambient','Jazz','Chill','Sleep']:h<9?['News','Classical','Folk','Talk']
    :h<12?['Top 40','Indie','Soul','Country']:h<18?['House','Funk','Latin','Pop']
    :h<22?['Classic Rock','Hip-Hop','Reggae','Blues']:['Techno','Ambient','Trance','Jazz'];
  const pool=[];
  const strongCC=VOCAB.countries.slice().sort((a,b)=>b[1]-a[1]).map(c=>c[0]);
  const known=COUNTRIES.map(c=>c[0]);
  const places=strongCC.filter(c=>known.includes(c)).slice(0,26);
  const rounds=[[],[],[]];
  moodFor.forEach(gn=>{ if(!GENRE[gn])return;
    const used=new Set();
    for(let i=0;i<3;i++){
      let cc=null,guard=0;
      while(guard++<12){ const c=places[Math.floor(Math.random()*places.length)];
        if(c&&!used.has(c)){cc=c;used.add(c);break;} }
      if(!cc)continue;
      rounds[i].push({genres:[GENRE[gn]],countries:[cc],regions:[],languages:[],excluded:[],ranking:null,
        label:gn+' in '+countryName(cc)});
    }
  });
  rounds.forEach(r=>r.forEach(d=>pool.push(d)));
  RANKINGS.slice(0,4).forEach(r=>{const gn=moodFor[Math.floor(Math.random()*moodFor.length)];
    if(GENRE[gn])pool.push({genres:[GENRE[gn]],countries:[],regions:[],languages:[],excluded:[],ranking:r.k,
      label:r.label+' '+gn.toLowerCase()});});
