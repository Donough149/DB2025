let trackTimer=null, lastRaw='', curTrack=null;
function parseTrack(raw,stationName){
  if(!raw)return null;
  let t=String(raw);
  try{ if(/%[0-9a-f]{2}/i.test(t)) t=decodeURIComponent(t.replace(/\+/g,' ')); }catch(e){}
  t=t.replace(/\s+/g,' ').trim();
  if(!t)return null;
  if(/^\{.*\}$/.test(t)){
    try{ const j=JSON.parse(t);
      const c=j.title||j.song||j.track||j.now_playing||'';
      if(!c)return null; t=String(c);
    }catch(e){return null;}
  }
  if(t.includes('~')){
    const f=t.split('~').map(x=>x.trim())
      .filter(x=>x&&!/^\d{4}$/.test(x)&&!/^\d{1,4}$/.test(x)
        &&!/^\d{4}-\d{2}-\d{2}T/.test(x)&&!/^[0-9a-f]{8}-[0-9a-f]{4}/i.test(x)
        &&!/^\.?\d+$/.test(x));
    if(f.length>=2)return validTrack({title:f[0],artist:f[1]},stationName);
    if(f.length===1)t=f[0];
  }
  t=t.replace(/\s*\|\s*[^|]*$/,'').trim();
  const xml=t.match(/<[^>]*ARTIST[^>]*>([^<]+)<\/[^>]+>[\s\S]*?<[^>]*TITLE[^>]*>([^<]+)</i);
  if(xml)return validTrack({artist:xml[1].trim(),title:xml[2].trim()},stationName);
  const mb=t.match(/text="([^"]+)"/i);
  if(mb)t=mb[1];
  if(/\*/.test(t)&&t.split('*').length>=3){
    const p=t.split('*').map(x=>x.trim()).filter(Boolean);
    if(/^(song|now playing)$/i.test(p[0]))p.shift();
    if(p.length>=2)return validTrack({title:p[0],artist:p[1]},stationName);
  }
  if(t.includes('|')){
    const p=t.split('|').map(x=>x.trim()).filter(Boolean);
    if(p.length>=3&&/^[\p{L}][\p{L}. '-]{2,}$/u.test(p[p.length-1]))
      return validTrack({title:p[0],artist:p[p.length-1]},stationName);
    t=p[0]||t;
  }
  if(/\.(mp3|aac|m4a|wav|ogg)\b/i.test(t))return null;
  if(/^[A-Z0-9]{3,6}(-FM|-AM)?\b/.test(t)&&/\d{2,3}[.,]\d\s*(FM|MHz)/i.test(t))return null;
  if(/^\s*\d{2,3}[.,]?\d?\s*(FM|AM|MHz)/i.test(t))return null;
  t=t.replace(/\s*[-–—]?\s*(https?:\/\/\S+|www\.\S+|\S+\.(com|net|fm|org|de|co\.uk))\s*$/i,'').trim();
  const debrand=x=>x.replace(/\s+[-–—]\s+[^-–—]*\b(radio|fm|on|live|vinyl|stream|hits|online)\b.*$/i,'').trim();
  const by=t.match(/^(.*?)\s+by\s+(.+)$/i);
  if(by&&by[1].length>1&&by[2].length>1)return validTrack({title:debrand(by[1]),artist:debrand(by[2])},stationName);
  const dash=t.split(/\s+[-–—]\s+/);
  if(dash.length>=2){
    const a=dash[0].trim(),b=dash.slice(1).join(' - ').trim();
    if(a&&b)return validTrack({artist:a,title:b},stationName);
  }
  const res=t.length>1?{title:t,artist:''}:null;
  return validTrack(res,stationName);
}
function validTrack(p,stationName){
  if(!p)return null;
  const norm=x=>fold(x||'').replace(/[^a-z0-9]/g,'');
  const title=(p.title||'').trim(), artist=(p.artist||'').trim();
  if(title.length<2)return null;
  if(/^https?:|^www\.|^tracklist\b/i.test(title))return null;
  if(!/[a-z0-9\p{L}]/iu.test(title))return null;
  if(stationName){
    const sn=norm(stationName), nt=norm(title+artist);
    if(sn.length>3&&nt.length>3&&(nt.includes(sn)||sn.includes(nt)))return null;
    const na=norm(artist);
    if(sn.length>3&&na.length>3&&(na===sn||sn.includes(na)||na.includes(sn)))return null;
  }
  if(artist&&norm(artist).length>2&&norm(artist)===norm(title))return null;
  if(/(^|\s)(www\.|https?:)|\.(com|net|org|fm|de|co\.uk|com\.au)(\s|$)/i.test(artist))return null;
  return {title,artist};
}
const ABC_SLUGS=[['triple j unearthed','unearthed'],['unearthed','unearthed'],
 ['double j','doublej'],['doublej','doublej'],['triple j','triplej'],['triplej','triplej'],
 ['abc jazz','jazz'],['abc classic','classic'],['abc country','country'],
 ['abc dig','dig'],['dig music','dig']];
function abcSlug(st){
  const n=((st&&st.name)||'').toLowerCase();
  if(!n||!/\b(abc|triple j|triplej|double j|doublej|unearthed|dig)\b/.test(n))return null;
  if(/news|radio national|\brn\b|grandstand|sport|conversations|local|extra/.test(n))return null;
  for(const [m,slug] of ABC_SLUGS) if(n.includes(m))return slug;
  return null;
}
async function abcNowPlaying(slug){
  try{
    const j=await fetch('https://music.abcradio.net.au/api/v1/plays/search.json?station='
      +enc(slug)+'&order=desc&limit=1',{cache:'no-store'}).then(r=>r.json());
    const rec=j&&j.items&&j.items[0]&&j.items[0].recording;
    if(!rec||!rec.title)return null;
    const artist=(rec.artists&&rec.artists[0]&&rec.artists[0].name)||'';
    return artist?(artist+' - '+rec.title):rec.title;
  }catch(e){return null;}
}
let npSource='', npAbort=null, npState='waiting';
function abortNP(){ if(npAbort){try{npAbort.abort();}catch(e){} npAbort=null;} }
let npDirectArt=null;
async function fetchNowPlaying(st,warm){
  const slug=abcSlug(st);
  if(slug){ const t=await abcNowPlaying(slug);
    if(t){ if(!warm)npSource='abc'; return {raw:t,art:null}; } }
  const ac=('AbortController' in window)?new AbortController():null;
  if(ac&&!warm)npAbort=ac;
  const t=ac?setTimeout(()=>ac.abort(),20000):null;
  const opt={cache:'no-store'}; if(ac)opt.signal=ac.signal;
  try{
    const r=await fetch('/np?u='+enc(streamOf(st)),opt);
    if(!r.ok){ if(!warm)npSource='none'; return {raw:null,art:null}; }
    const j=await r.json();
    const raw=(j&&j.raw)||null, art=(j&&j.art)||null;
    if(!warm){ npDirectArt=art; npSource=raw?'phone':'none'; }
    return {raw,art};
  }catch(e){ if(!warm)npSource='none'; return {raw:null,art:null}; }
  finally{ if(t)clearTimeout(t); if(npAbort===ac)npAbort=null; }
}
async function artFor(q){
  try{
    const j=await fetch('https://itunes.apple.com/search?limit=1&entity=song&term='+enc(q)).then(r=>r.json());
    const a=j.results&&j.results[0]; if(!a)return null;
    return {art:(a.artworkUrl100||'').replace('100x100bb','600x600bb'),
            artist:a.artistName,title:a.trackName};
  }catch(e){return null;}
}
const coverCache={};
function generateCover(st){
  const key='gen:'+(st.stationuuid||st.name);
  if(coverCache[key])return coverCache[key];
  const N=600,c=document.createElement('canvas');c.width=c.height=N;
  const x=c.getContext('2d');
  const h=hashHue(st.name||'?');
  const a=hsv2rgb(h,.62,.62), b=hsv2rgb((h+42)%360,.70,.34);
  const g=x.createLinearGradient(0,0,N,N);
  g.addColorStop(0,'rgb('+a.join(',')+')'); g.addColorStop(1,'rgb('+b.join(',')+')');
  x.fillStyle=g; x.fillRect(0,0,N,N);
  const rg=x.createRadialGradient(N*.32,N*.28,10,N*.32,N*.28,N*.9);
  rg.addColorStop(0,'rgba(255,255,255,.20)'); rg.addColorStop(1,'rgba(255,255,255,0)');
  x.fillStyle=rg; x.fillRect(0,0,N,N);
  x.strokeStyle='rgba(255,255,255,.10)'; x.lineWidth=2;
  for(let i=1;i<=4;i++){x.beginPath();x.arc(N*.5,N*.5,i*62,0,Math.PI*2);x.stroke();}
  const words=(st.name||'?').replace(/[^\p{L}\p{N} ]/gu,' ').trim().split(/\s+/).filter(Boolean);
  const mono=(words.slice(0,2).map(w=>w[0]).join('')||'♪').toUpperCase();
  x.fillStyle='rgba(255,255,255,.92)';
  x.font='600 '+(mono.length>1?190:230)+'px -apple-system,Segoe UI,Roboto,sans-serif';
  x.textAlign='center'; x.textBaseline='middle';
  x.fillText(mono,N*.5,N*.53);
  const url=c.toDataURL('image/png');
  coverCache[key]=url; return url;
}
async function huntArtwork(st){
  if(st.favicon)return st.favicon;
  const key='hunt:'+st.stationuuid;
  if(coverCache[key]!==undefined)return coverCache[key]||generateCover(st);
  let found=null;
  try{
    const twins=await api('/stations/byname/'+enc(st.name));
    const t=(twins||[]).find(x=>x.favicon&&stem(x.name)===stem(st.name));
    if(t)found=t.favicon;
  }catch(e){}
  if(!found&&st.homepage){
    try{
      const j=await fetch('/art?u='+enc(st.homepage),{cache:'no-store'}).then(r=>r.json());
      if(j&&j.art)found=j.art;
    }catch(e){}
  }
  coverCache[key]=found;
  return found||generateCover(st);
}
let artLayer=0, artSeq=0, artPending='', artShownUrl='';
function proxied(u){
  return (/^https?:/i.test(u)&&u.indexOf(location.host)<0) ? '/img?u='+enc(u) : u;
}
function showArt(url){
  if(!url)return;
  const a=$('bigLogo'),b=$('bigLogoB');
  if(artShownUrl===url||artPending===url)return;
  artPending=url;
  const my=++artSeq;
  const next=artLayer?a:b, prev=artLayer?b:a;
  artLayer=artLayer?0:1;
  const img=new Image(); const srcURL=url;
  const done=()=>{ if(artPending===url)artPending=''; };
  img.onload=()=>{
    done();
    if(my!==artSeq)return;
    next.src=url; next.dataset.src=url; artShownUrl=url;
    $('bigLogoFallback').style.opacity='0';
    const commit=()=>{ if(my!==artSeq)return;
      next.classList.remove('out'); next.classList.add('show');
      prev.classList.remove('show'); prev.classList.add('out'); };
    requestAnimationFrame(commit); setTimeout(commit,120);
    ambientFromURL(url); setArtBackground(url);
  };
  img.onerror=()=>{
    if(!img._retried&&proxied(url)!==url){
      img._retried=1; img.src=proxied(url); return;
    }
    done(); if(my!==artSeq)return;
    if(cur){const g=generateCover(cur); if(g!==url&&g!==artShownUrl)showArt(g);} };
  img.src=srcURL;
}
function showTrack(title,artist,art){

        nm.style.transform='none';ar.style.transform='none';});

  setTimeout(solveNowPlaying,300);

        title: title||niceName(cur), artist: artist||subOf(cur), album: niceName(cur),
        artwork: art?[{src:art,sizes:'600x600',type:'image/jpeg'}]
                    :(cur.favicon?[{src:cur.favicon}]:[])});
    }catch(e){}
  }
}
let bgLayer=0;
function setArtBackground(url){
  const a=$('artBg'),b=$('artBgB'); if(!a||!b)return;
  if(!url){a.classList.remove('on');b.classList.remove('on');
    a.style.backgroundImage='';b.style.backgroundImage='';return;}
  const css='url("'+url.replace(/"/g,'%22')+'")';
  const next=bgLayer?a:b, prev=bgLayer?b:a;
  if(next.style.backgroundImage===css||prev.style.backgroundImage===css)return;
  next.style.backgroundImage=css;
  const commit=()=>{next.classList.add('on');prev.classList.remove('on');};
  requestAnimationFrame(commit); setTimeout(commit,120);
  bgLayer=bgLayer?0:1;
}
function ambientFromURL(url){
  const img=new Image(); img.crossOrigin='anonymous';
  img.onload=()=>{ try{
    const w=28,c=document.createElement('canvas');c.width=c.height=w;
    const x=c.getContext('2d');x.drawImage(img,0,0,w,w);
    const d=x.getImageData(0,0,w,w).data;
    const bins=new Array(24).fill(0),bh=new Array(24).fill(0),bs=new Array(24).fill(0),bv=new Array(24).fill(0);
    for(let i=0;i<d.length;i+=4){
      const r=d[i]/255,g=d[i+1]/255,b=d[i+2]/255;
      const mx=Math.max(r,g,b),mn=Math.min(r,g,b),v=mx,sst=mx?(mx-mn)/mx:0;
      if(v<.12||sst<.22||(v>.92&&sst<.35))continue;
      let h=0;const df=mx-mn;
      if(df){if(mx===r)h=((g-b)/df)%6;else if(mx===g)h=(b-r)/df+2;else h=(r-g)/df+4;h*=60;if(h<0)h+=360;}
      const bin=Math.floor(h/15)%24,wg=sst*v;bins[bin]+=wg;bh[bin]+=h*wg;bs[bin]+=sst*wg;bv[bin]+=v*wg;
    }
    const order=bins.map((v,i)=>[v,i]).sort((a,b)=>b[0]-a[0]).filter(x=>x[0]>0);
    if(!order.length)return;
    const mk=i=>{const h=bh[i]/bins[i];
      const sat=Math.max(bs[i]/bins[i],.55),val=Math.min(Math.max(bv[i]/bins[i],.5),.9);
      return hsv2rgb(h,sat,val);};
    const primary=mk(order[0][1]);
    const far=order.find(([,i])=>Math.min(Math.abs(i-order[0][1]),24-Math.abs(i-order[0][1]))>=3);
    setAmbient(primary,!au.paused, far?mk(far[1]):null);
  }catch(e){} };
  img.src=proxied(url);
}
async function pollTrack(st){
  if(!cur||cur.stationuuid!==st.stationuuid)return;
  lastPollAt=Date.now();
  const raw=(await fetchNowPlaying(st)).raw;
  if(!cur||cur.stationuuid!==st.stationuuid)return;
  if(!raw){ if(!curTrack){npState='none'; renderNP();} return; }
  if(raw===lastRaw)return;
  lastRaw=raw;
  const p=parseTrack(raw,st.name);
  if(!p){ if(!curTrack){npState='junk'; renderNP();} return; }
  curTrack={title:p.title,artist:p.artist};
  npState='track';
  showTrack(p.title,p.artist);
  if(npDirectArt){
    showTrack(p.title,p.artist,npDirectArt); return;
  }
  const q=[p.artist,p.title].filter(Boolean).join(' ');
  const found=await artFor(q);
  if(found&&cur&&cur.stationuuid===st.stationuuid&&lastRaw===raw){
    curTrack={title:found.title,artist:found.artist};
    showTrack(found.title,found.artist,found.art);
  }
}
let trackWatchFor=null, lastPollAt=0;
function startTrackWatch(st){
  clearInterval(trackTimer); abortNP();
  trackWatchFor=st&&st.stationuuid; npState='waiting';
  lastRaw=''; curTrack=null; let tries=0;
  const tick=async()=>{
    if(!cur||cur.stationuuid!==st.stationuuid)return;
    await pollTrack(st); tries++;
    if(!lastRaw&&tries<4)setTimeout(tick,tries*4000);
  };
  tick();
  trackTimer=setInterval(()=>pollTrack(st),18000);
}

const WCODE={0:['☀️','Clear sky'],1:['🌤','Mainly clear'],2:['⛅️','Partly cloudy'],3:['☁️','Overcast'],
 45:['🌫','Fog'],48:['🌫','Rime fog'],51:['🌦','Light drizzle'],53:['🌦','Drizzle'],55:['🌧','Heavy drizzle'],
 61:['🌦','Light rain'],63:['🌧','Rain'],65:['🌧','Heavy rain'],71:['🌨','Light snow'],73:['🌨','Snow'],
 75:['❄️','Heavy snow'],80:['🌦','Showers'],81:['🌧','Showers'],82:['⛈','Violent showers'],
 95:['⛈','Thunderstorm'],96:['⛈','Thunderstorm'],99:['⛈','Thunderstorm']};
const wxCache={};
async function fetchWeather(st){
  if(!st)return null;
  const cands=[st.state,CNAME[st.countrycode]?CNAME[st.countrycode].name:countryName(st.countrycode)]
    .map(x=>(x||'').trim()).filter(x=>x.length>1);
  const key=cands.join('|'); if(!key)return null;
  if(wxCache[key]!==undefined)return wxCache[key];
  for(const c of cands){
    try{
      const g=await fetch('https://geocoding-api.open-meteo.com/v1/search?count=1&language=en&name='+enc(c)).then(r=>r.json());
      const hit=g&&g.results&&g.results[0]; if(!hit)continue;
      const w=await fetch('https://api.open-meteo.com/v1/forecast?latitude='+hit.latitude
        +'&longitude='+hit.longitude+'&current=temperature_2m,weather_code&timezone=auto').then(r=>r.json());
      if(!w||!w.current)continue;
      const out={temp:Math.round(w.current.temperature_2m),code:w.current.weather_code,
        tz:w.timezone,place:hit.name};
      wxCache[key]=out; return out;
    }catch(e){}
  }
  wxCache[key]=null; return null;
}
let wxTimer=null;
async function renderWeather(st){
  const el=$('npWeather'); if(!el)return;
  el.textContent='';
  clearInterval(wxTimer);
  const wx=await fetchWeather(st);
  if(!wx||!cur||cur.stationuuid!==st.stationuuid){return;}
  const paint=()=>{
    const t=new Date().toLocaleTimeString('en-GB',{timeZone:wx.tz,hour:'numeric',minute:'2-digit'});
    const [emoji,desc]=WCODE[wx.code]||['',''];
    el.textContent=[emoji+' '+desc,wx.temp+'°C',t,wx.place].filter(Boolean).join(' · ');
  };
  paint(); wxTimer=setInterval(paint,30000);
}
